cmake_minimum_required(VERSION 3.21)

project(tasks LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find required packages
find_package(Qt6 REQUIRED COMPONENTS Core Test)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)

# Source files
set(SOURCES
    Task.cpp
    TaskScheduler.cpp
    ImageProcessingTask.cpp
)

set(HEADERS
    Task.h
    TaskScheduler.h
    ImageProcessingTask.h
)

# Create library
add_library(${PROJECT_NAME} STATIC
    ${SOURCES}
    ${HEADERS}
)

# Include directories
target_include_directories(${PROJECT_NAME}
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/../image_processing/core
        ${CMAKE_CURRENT_SOURCE_DIR}/../image_processing/operations
)

# Link libraries
target_link_libraries(${PROJECT_NAME}
    PUBLIC
        Qt6::Core
        Qt6::Test
        opencv_core
        opencv_imgproc
    PRIVATE
        image_processing
)

# Avoid Qt keyword conflicts with Halide (Qt defines 'emit' as a macro)
target_compile_definitions(${PROJECT_NAME} PRIVATE QT_NO_KEYWORDS)

# Compiler options
target_compile_options(${PROJECT_NAME} PRIVATE
    $<$<CONFIG:Debug>:-Wall -Wextra -Wpedantic -g>
    $<$<CONFIG:Release>:-O3>
)

# Enable Qt MOC
set_target_properties(${PROJECT_NAME} PROPERTIES
    AUTOMOC ON
)

# Testing (optional)
if(BUILD_TESTING)
    enable_testing()
    add_subdirectory(test)
endif()
